=== FB_RealCylinderGroup ===
--- DECL ---

--- IMPL ---
(* 模块说明：按料型选通的真实气缸组适配层（G2 侧挡用，300/450/600 三缸）。
   同一时刻只允许一个气缸下降，其余强制提升；回写与 FB_SimDevice 完全一致的接收/完成序号，
   使上层流程与互锁无需区分单缸、多缸或仿真。
   命令目标 lrTarget：0 = 三缸全部提升；300 / 450 / 600 = 对应料型气缸下降。
   lrTarget 在 0..1 之间时按单缸旧语义映射：>0.5 视为「降到当前料型」，<=0.5 视为全部提升。 *)
VAR_INPUT
	xUseReal : BOOL; (* TRUE 表示本编号由真实气缸组接管；FALSE 时三个阀全部失电且不响应任何命令。 *)
	xEnable : BOOL; (* 设备级使能，来自整机许可；FALSE 时阀失电并清序号。 *)
	xReset : BOOL; (* 故障复位上升沿；清锁存错误。 *)
	xFaultHold : BOOL := FALSE; (* 整机故障保持：TRUE 时三个阀立即失电，直到复位。 *)
	uiMaterialModel : UINT; (* 当前运行料型 300/450/600；单缸旧语义 lrTarget>0.5 时降到本料型。 *)
	uiBand300 : UINT := 300; (* 料型 1 的带号，与 axG2RelayDemand[1] 对应。 *)
	uiBand450 : UINT := 450; (* 料型 2 的带号。 *)
	uiBand600 : UINT := 600; (* 料型 3 的带号。 *)
	xRetract1 : BOOL; (* 1 号（300）气缸提升到位传感器原始电平。 *)
	xExtend1 : BOOL; (* 1 号（300）气缸下降到位传感器原始电平。 *)
	xRetract2 : BOOL; (* 2 号（450）气缸提升到位传感器原始电平。 *)
	xExtend2 : BOOL; (* 2 号（450）气缸下降到位传感器原始电平。 *)
	xRetract3 : BOOL; (* 3 号（600）气缸提升到位传感器原始电平。 *)
	xExtend3 : BOOL; (* 3 号（600）气缸下降到位传感器原始电平。 *)
	xInvertSensors : BOOL; (* TRUE 表示六个传感器均为低电平有效，本层统一取反。 *)
	xInvertValve : BOOL; (* TRUE 表示三个阀均为低电平使能。 *)
	tSensorTimeout : TIME := T#3S; (* 任一端两端均无信号达到该时间判为传感器丢失，报 3004。 *)
	tTravelTimeout : TIME := T#10S; (* 命令发出后目标缸未到位达到该时间判为动作超时，报 3005。 *)
	stConfig : ST_DeviceConfig; (* 行程范围，语义与仿真侧一致；组内气缸只使用 0/1 两端。 *)
	stCommand : ST_DeviceCommand; (* 上层统一设备命令。 *)
END_VAR
VAR_OUTPUT
	stStatus : ST_DeviceStatus; (* 与 FB_SimDevice 语义一致；lrPosition 目标缸下降到位=1.0，全部提升=0.0。 *)
	aValve : ARRAY[1..3] OF BOOL; (* 三个阀的电气电平（已含 xInvertValve），直接接物理输出点。 *)
	uiActiveBand : UINT; (* 当前下降的料型带号；0 表示三个缸都在提升位。 *)
END_VAR
VAR
	rtReset : R_TRIG; (* 复位上升沿。 *)
	tonNoSensor : TON; (* 任一端两端均无信号的持续计时。 *)
	tonTravel : TON; (* 命令发出后等待到位的计时。 *)
	udiLastID : UDINT; (* 上次已接受的命令序号。 *)
	aRetract : ARRAY[1..3] OF BOOL; (* 取反处理后的三路提升信号。 *)
	aExtend : ARRAY[1..3] OF BOOL; (* 取反处理后的三路下降信号。 *)
	xCmdValid : BOOL; (* 当前存在有效的保位命令，阀按此持续输出。 *)
	uiTargetBand : UINT; (* 有效命令期望下降的料型带号；0 表示三缸全部提升。 *)
	xTravelling : BOOL; (* 尚未确认到达命令要求的状态。 *)
	xAnyRetract : BOOL; (* 三个提升信号中至少一个有效。 *)
	xAnyExtend : BOOL; (* 三个下降信号中至少一个有效。 *)
	xSensorFault : BOOL; (* 传感器丢失或两端同时有效。 *)
	uiFaultBand : UINT; (* 出问题的气缸序号 1..3，供错误定位。 *)
	i : INT; (* 本扫描数组遍历索引；含义及范围以当前 FOR 循环为准。 *)
END_VAR
rtReset(CLK := xReset);

(* ===== 1. 复位：只清错误与序号，姿态仍由传感器决定 ===== *)
IF xUseReal AND rtReset.Q THEN
	stStatus.xError := FALSE; stStatus.udiErrorID := 0;
	xCmdValid := FALSE; xTravelling := FALSE; udiLastID := 0; uiTargetBand := 0;
	stStatus.udiAcceptedID := 0; stStatus.udiCompletedID := 0;
	tonNoSensor(IN := FALSE);
	tonTravel(IN := FALSE);
END_IF

(* ===== 2. 传感器取反与逐缸姿态判定 ===== *)
aRetract[1] := xRetract1 XOR xInvertSensors;
aExtend[1] := xExtend1 XOR xInvertSensors;
aRetract[2] := xRetract2 XOR xInvertSensors;
aExtend[2] := xExtend2 XOR xInvertSensors;
aRetract[3] := xRetract3 XOR xInvertSensors;
aExtend[3] := xExtend3 XOR xInvertSensors;

(* 任一端两端同时有效：传感器短路、磁环位置错或机械卡在中间，姿态不可信 *)
xSensorFault := FALSE;
uiFaultBand := 0;
FOR i := 1 TO 3 DO
	IF aRetract[i] AND aExtend[i] THEN
		xSensorFault := TRUE;
		IF uiFaultBand = 0 THEN uiFaultBand := INT_TO_UINT(i); END_IF
	END_IF
END_FOR

(* 任一端两端都没有信号并持续 tSensorTimeout：判传感器丢失；全部气缸同时离线时不报，避免整组掉线刷屏 *)
xAnyRetract := aRetract[1] OR aRetract[2] OR aRetract[3];
xAnyExtend := aExtend[1] OR aExtend[2] OR aExtend[3];
tonNoSensor(IN := xUseReal AND NOT xSensorFault AND NOT xAnyRetract AND NOT xAnyExtend, PT := tSensorTimeout);
IF xSensorFault THEN
	stStatus.xError := TRUE; stStatus.udiErrorID := 3003;
ELSIF tonNoSensor.Q AND NOT stStatus.xError THEN
	stStatus.xError := TRUE; stStatus.udiErrorID := 3004;
END_IF

(* ===== 3. 命令接收与保位判定 ===== *)
IF NOT xUseReal THEN
	(* 未接管：不驱动阀，也不发布就绪，避免被上层当成可用设备 *)
	xCmdValid := FALSE; xTravelling := FALSE; udiLastID := 0; uiTargetBand := 0;
	stStatus.xReady := FALSE; stStatus.xBusy := FALSE; stStatus.xActive := FALSE;
	stStatus.udiAcceptedID := 0; stStatus.udiCompletedID := 0;
ELSIF NOT xEnable OR stStatus.xError OR xFaultHold THEN
	(* 故障或整机故障保持时阀失电，不带错误继续动作 *)
	xCmdValid := FALSE; xTravelling := FALSE;
	stStatus.xReady := FALSE; stStatus.xBusy := FALSE; stStatus.xActive := FALSE;
ELSE
	stStatus.xReady := xEnable;

	(* 新序号即新命令；带号未知时只接受「全部提升」，防止朝未知料型动作 *)
	IF (stCommand.udiID <> 0) AND (stCommand.udiID <> udiLastID)
		AND xEnable AND NOT stStatus.xBusy THEN
		udiLastID := stCommand.udiID;
		stStatus.udiAcceptedID := udiLastID; stStatus.udiCompletedID := 0;
		CASE stCommand.eAction OF
		E_DeviceAction.Position:
			IF (stCommand.lrTarget > 0.5) AND (stCommand.lrTarget <= 1.0) THEN
				(* 单缸旧语义：1 = 降到当前料型 *)
				uiTargetBand := 0;
				IF (uiMaterialModel = uiBand300) OR (uiMaterialModel = uiBand450)
					OR (uiMaterialModel = uiBand600) THEN
					uiTargetBand := uiMaterialModel;
				ELSE
					stStatus.xError := TRUE; stStatus.udiErrorID := 2001;
				END_IF
			ELSIF ABS(stCommand.lrTarget) < 0.01 THEN
				uiTargetBand := 0;
			ELSIF (stCommand.lrTarget = uiBand300) OR (stCommand.lrTarget = uiBand450)
				OR (stCommand.lrTarget = uiBand600) THEN
				uiTargetBand := LREAL_TO_UINT(stCommand.lrTarget);
			ELSE
				(* 只支持 0 / 300 / 450 / 600，其它目标明确报错而不是静默不动 *)
				stStatus.xError := TRUE; stStatus.udiErrorID := 2001;
			END_IF;
			IF NOT stStatus.xError THEN
				xCmdValid := TRUE;
				xTravelling := TRUE;
				stStatus.xBusy := TRUE;
			ELSE
				xCmdValid := FALSE;
			END_IF
		E_DeviceAction.Stop, E_DeviceAction.None:
			xCmdValid := FALSE; xTravelling := FALSE; uiTargetBand := 0;
			stStatus.udiAcceptedID := 0; stStatus.udiCompletedID := 0;
			udiLastID := 0;
		ELSE
			(* 气缸组没有连续运行、点动和回零，明确报错而不是静默不动 *)
			stStatus.xError := TRUE; stStatus.udiErrorID := 2001;
			xCmdValid := FALSE;
		END_CASE
	END_IF

	(* 命令被撤销或失能：立即取消保位，阀随之失电 *)
	IF (stCommand.eAction = E_DeviceAction.Stop) OR (stCommand.eAction = E_DeviceAction.None)
		OR NOT xEnable THEN
		xCmdValid := FALSE; xTravelling := FALSE;
		stStatus.xBusy := FALSE;
	END_IF

	(* 到位判定：目标缸下降到位，且另外两个缸都在提升位，才算本次命令完成 *)
	IF xCmdValid THEN
		IF uiTargetBand = 0 THEN
			IF aRetract[1] AND aRetract[2] AND aRetract[3] THEN
				IF xTravelling THEN stStatus.udiCompletedID := udiLastID; END_IF
				xTravelling := FALSE;
				stStatus.xBusy := FALSE;
			END_IF
		ELSIF (uiTargetBand = uiBand300) THEN
			IF aExtend[1] AND aRetract[2] AND aRetract[3] THEN
				IF xTravelling THEN stStatus.udiCompletedID := udiLastID; END_IF
				xTravelling := FALSE;
				stStatus.xBusy := FALSE;
			END_IF
		ELSIF (uiTargetBand = uiBand450) THEN
			IF aExtend[2] AND aRetract[1] AND aRetract[3] THEN
				IF xTravelling THEN stStatus.udiCompletedID := udiLastID; END_IF
				xTravelling := FALSE;
				stStatus.xBusy := FALSE;
			END_IF
		ELSIF (uiTargetBand = uiBand600) THEN
			IF aExtend[3] AND aRetract[1] AND aRetract[2] THEN
				IF xTravelling THEN stStatus.udiCompletedID := udiLastID; END_IF
				xTravelling := FALSE;
				stStatus.xBusy := FALSE;
			END_IF
		END_IF
	END_IF
END_IF

(* ===== 4. 位置反馈：目标缸下降到位=1.0，全部提升=0.0，中间保持上次值 ===== *)
IF uiTargetBand = uiBand300 THEN
	IF aExtend[1] THEN stStatus.lrPosition := 1.0; END_IF
ELSIF uiTargetBand = uiBand450 THEN
	IF aExtend[2] THEN stStatus.lrPosition := 1.0; END_IF
ELSIF uiTargetBand = uiBand600 THEN
	IF aExtend[3] THEN stStatus.lrPosition := 1.0; END_IF
ELSIF aRetract[1] AND aRetract[2] AND aRetract[3] THEN
	stStatus.lrPosition := 0.0;
END_IF
uiActiveBand := 0;
IF (uiTargetBand = uiBand300) AND aExtend[1] THEN uiActiveBand := uiBand300; END_IF
IF (uiTargetBand = uiBand450) AND aExtend[2] THEN uiActiveBand := uiBand450; END_IF
IF (uiTargetBand = uiBand600) AND aExtend[3] THEN uiActiveBand := uiBand600; END_IF

(* ===== 5. 到位超时 ===== *)
tonTravel(IN := xUseReal AND xTravelling AND xCmdValid, PT := tTravelTimeout);
IF tonTravel.Q AND NOT stStatus.xError THEN
	stStatus.xError := TRUE; stStatus.udiErrorID := 3005;
	stStatus.xBusy := FALSE;
	xTravelling := FALSE;
END_IF

(* ===== 6. 阀输出：按保位命令持续输出，只允许一个缸下降 ===== *)
aValve[1] := FALSE; aValve[2] := FALSE; aValve[3] := FALSE;
IF xCmdValid AND xUseReal AND NOT stStatus.xError AND NOT xFaultHold THEN
	aValve[1] := (uiTargetBand = uiBand300);
	aValve[2] := (uiTargetBand = uiBand450);
	aValve[3] := (uiTargetBand = uiBand600);
END_IF
IF xInvertValve THEN
	aValve[1] := NOT aValve[1]; aValve[2] := NOT aValve[2]; aValve[3] := NOT aValve[3];
END_IF

(* ===== 7. 收尾 ===== *)
stStatus.xReady := stStatus.xReady AND xUseReal AND NOT stStatus.xError;
stStatus.xActive := xCmdValid;
(* 气缸组没有回零动作，用“任一端姿态已由传感器确立”代替 *)
stStatus.xHomed := xUseReal AND (xAnyRetract OR xAnyExtend);
